\begin{enumerate}
	
		%ondas banda ap
		\item La perturbación asociada a una onda viene descrita por la expresión
$$\psi(z,t)= 10^{-3}\sin\left(10t +2\pi z-\frac{3\pi}{2}\right),\; \text{\small (SI)}.$$
	\begin{itemize}
		\item[a)] Determina: \textbf{el sentido y la dirección de propagación, la longitud de onda y la velocidad de propagación}.
		\begin{mdframed}[backgroundcolor=yellow!20]  
		La dirección es el eje $z$, recorrida en sentido negativo por el cambio de signo entre la parte espacial y temporal de la fase. Leyendo directamente $k=2\pi\,m^{-1}$ y $\omega=10\,s^{-1}$ de $\psi(z,t)$ y  a partir de las propiedades de periocidad espacial de la onda podemos definir la longitud de onda como	\(k=\frac{2\pi}{\lambda}\rightarrow \lambda=2\pi k^{-1}\simeq 1\,m.\) Usando la relación de dispersión podemos calcular $c=\omega k^{-1}\simeq \sfrac{10}{2\pi}\simeq 1.6\,ms^{-1}.$
		\end{mdframed}
		\item[b)] Obtén la \textbf{diferencia de fase}, en el mismo instante de tiempo, para dos puntos separados una distancia igual a media longitud de onda.
		\begin{mdframed}[backgroundcolor=yellow!20]  
		Si $\Delta t\equiv 0$ entonces la diferencia de fase $\Delta\phi=k\Delta z+\omega \Delta t$ queda $\Delta\phi\big|_{\Delta t=0}=k\Delta z$. Para $\Delta z=\sfrac{\lambda}{2}$ esta cantidad es igual a $\Delta\phi\big|_{\Delta t=0}=\sfrac{k\lambda}{2}=\pi.$
		\end{mdframed}
	\end{itemize}
	\qed
	\vspace{2ex}

			%ondas banda not-sob.
	\item 	Un búho que se encuentra en un árbol a una altura de $30\,m$ emite continuamente una onda sonora propagando una potencia  de $40\;nW$. Si un ratón se acerca a las proximidades del árbol, determina:
			\begin{itemize}
				\item[a)] La \textbf{distancia al pie del árbol} a la que el ratón comienza a oír al búho.
           	\begin{mdframed}[backgroundcolor=yellow!20] 
            	Suponiendo que el umbral auditivo del ratón es similar al del oído humano, y considerando que se cumplen las hipótesis necesarias para suponer que la onda sonora es aproximadamente esférica y sin atenuación, en el punto umbral en el que el el búho comienza a ser oído se cumplirá
            	$$I(r)=I_0\longrightarrow \frac{P}{4\pi r^2}=I_0,$$
            	donde $r$ es la distancia radial del foco sonoro a la posición del ratón. 
            	
            	Naturalmente, al estar encaramado a un árbol de altura $h$, se cumplirá
            	$$h^2+d^2=r^2,$$
            	con $d$ la distancia del ratón al pie del árbol. Con todo
            	$$\frac{P}{4\pi(h^2+d^2)}=I_0\longrightarrow d=\sqrt{\frac{P}{4\pi I_0}-h^2}\simeq 47.8\;m.$$
            	\end{mdframed}
				\item[b)] El \textbf{nivel sonoro} percibido por el ratón \textit{cuando esté junto al árbol}.
			\end{itemize}
			\vspace{0ex}
			
		 \begin{footnotesize} 
            	\textbf{Indicación}: Supón que para el ratón sigue valiendo $I_0\equiv 10^{-12}\;W\,m^{-2}$.			\end{footnotesize}
            	
            	\begin{mdframed}[backgroundcolor=yellow!20] 
            	En el pie del árbol, $r=h$, de manera que lo que nos piden es
            	$$B(r=d)=10\log_{10}\left(\frac{P}{4\pi h^2I_0}\right)\simeq 5.5\;dB.$$
            	\end{mdframed}
            	
            	
		\qed

		%sonido banda ap.
		\item 
			\begin{itemize}
				\item[a)] El sonido emitido por un altavoz es registrado con un nivel sonoro de $50\;dB$ a una 
distancia de $3\;m$. Si el altavoz se aproxima como un foco puntual, estima la \textbf{distancia} a la que el sonido deja de ser perceptible para un ser humano.
  \begin{mdframed}[backgroundcolor=yellow!20]  
  A una distancia $d$ del foco el nivel sonoro es $B(d)=10\log_{10}\big[P(4\pi d^{-2}I_0)^{-1}\big]$, de modo que $P=4\pi\,d^2\,I_0\;10^{\sfrac{B(d)}{10}},$ que se conserva aproximadamente en la propagación.  Cuando el sonido deje de ser percibido a una distancia $r$ su intensidad igualará la umbral y se cumplirá $I(r)=I_0$, es decir
  \[\frac{P}{4\pi r^2}=I_0\rightarrow 4\pi\,d^2\,I_0\;10^{\sfrac{B(d)}{10}}=I_0\,4\pi r^2\rightarrow d^210^{\sfrac{B(d)}{10}}=r^2. \]
  Finalmente $r=d\,10^{\sfrac{B(d)}{20}}\simeq 3\cdot 10^{\sfrac{50}{20}}\simeq 1\;km$.
  \end{mdframed}
				\item[b)] Una persona hablando produce un sonido con una intensidad unas seiscientas veces mayor que 						cuando susurra. ¿Qué \textbf{diferencia de nivel sonoro} hay entre ambos sonidos?
				\vspace*{.2mm}					
			\end{itemize}
			\begin{footnotesize} 
            	\textbf{Dato}: \hspace*{2ex} $\triangleright\;I_0\equiv 10^{-12}\;W\,m^{-2}$.			\end{footnotesize}
       \begin{mdframed}[backgroundcolor=yellow!20]  
        	Si llamamos $I$ a la intensidad asociada al susurro, la intensidad asociada al habla es $600I$. La diferencia en el nivel sonoro será
        	\[\Delta B=10\log600=10(\log 6+2)\simeq 27.8\;dB.\]
       \end{mdframed}
		\qed

		%banda notable

		\item 	Un láser aproximadamente monocromático emite radiación electromagnética luminosa de $405\;nm$ trabajando a una potencia de $20\;W$. 
			\begin{itemize}
				\item[a)] Suponiendo que la radiación se propaga en el vacío, determina su \textbf{frecuencia} y el \textbf{número de fotones que emite por segundo}.
				\begin{mdframed}[backgroundcolor=yellow!20]  
				A partir de la relación de dispersión y sabiendo que en el vacío la luz se propaga a una velocidad $c_0$ encontramos ${f=\sfrac{c_0}{\lambda_0}\simeq 7.4\cdot 10^{14}\;s^{-1}.}$ 
				
				En virtud de la ley de Planck, un conjunto de $N$ fotones que propaga, por intervalo de tiempo $\Delta t$, una potencia $P=\frac{\Delta E}{\Delta t}=\frac{Nhf}{\Delta t},$	de modo que la cantidad de fotones emitidos por unidad de tiempo será
				$${\frac{N}{\Delta t}=\frac{P}{hf}.}$$
				\end{mdframed}
				\item[b)] Si en el transcurso de su trayectoria el haz láser ingresa en un medio homogéneo e isótropo cuyo índice de refracción es $n=1.5$; calcula la \textbf{potencia del haz} y su \textbf{longitud de onda} en este nuevo medio.
			\end{itemize}
			\begin{footnotesize} 
            	\textbf{Datos}: \hspace*{2ex} $\triangleright\;c_0\simeq 3\cdot 10^8\;m\,s^{-1}\hspace{1cm}\triangleright\;h\simeq 6.63\cdot 10^{-34}\;J\,s.$			
            \end{footnotesize}
            
            \begin{mdframed}[backgroundcolor=yellow!20]  
				La potencia no varía si despreciamos fenómenos de absorción en el material. Imponiendo esta condición a la relación de dispersión en ambos medios resulta
				$$\Delta f\bigg|_{0\rightarrow 1}\!\!=0\longrightarrow\frac{c_0}{\lambda_0}=\frac{c_1}{\lambda_1}\longrightarrow {\lambda_1=\frac{c_1}{c_0}\lambda_0=\frac{n_0}{n_1}\lambda_0\simeq 270\;nm.}$$
				\end{mdframed}
				
			\qed
	\vspace{2ex}

\item Una superficie plana separa dos medios transparentes de índices de refracción $n_1
=2$ y $n_2 =1.4$ respectivamente. Un rayo luminoso incide sobre la frontera propagándose hacia el medio con menor índice  de forma tal que el rayo reflejado y el transmitido \textit{son perpendiculares entre sí}.  Calcula el valor  \textbf{a)} del \textbf{ángulo de incidencia} y \textbf{b)} del \textbf{ángulo de transmisión}.
	
	\begin{mdframed}[backgroundcolor=yellow!20] 
		Se ha de cumplir
		$$\alpha_r+90+\alpha_t=180\longrightarrow \alpha_r+\alpha_t=90,$$
		es decir, el ángulo transmitido y el reflejado son complementarios. 
		
		Por otro lado la primera ley de Snell nos indica que $\alpha_i=\alpha_r,$ de donde $\alpha_i+\alpha_t=90.$
		Como los rayos incidente y transmitido deben cumplir la segunda ley de Snell resulta
		$$n_1\sin\alpha_i=n_2\sin\alpha_t\longrightarrow n_1\sin\alpha_i=n_2\sin(90-\alpha_i)=n_2\cos\alpha_i,$$
		de donde se deduce que $\tan\alpha_i=\sfrac{n_2}{n_1}\rightarrow{\alpha_i=\arctan{0.7}\simeq 35^\circ.}$ Una vez calculado este valor es fácil hallar $\alpha_t\simeq 55^\circ.$
		\end{mdframed}
\qed
